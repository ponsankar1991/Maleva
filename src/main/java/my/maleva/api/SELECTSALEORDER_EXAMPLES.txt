// EXAMPLE 11: Combined Complex Filter
// ============================================================================
REQUEST:
POST /api/sale-orders/search
Content-Type: application/json

{
    "Comid": 1,
    "Id": 5,
    "JId": 2,
    "Employeeid": 3,
    "DashboardStatus": 1,
    "statusList": "1,2",
    "Statusid": 0,
    "completestatusnotshow": true,
    "Remarks": 1,
    "Offvesselname": "Maersk",
    "Loadingvesselname": null,
    "Search": null,
    "Invoice": false,
    "ETA": true,
    "ETAType": 2,
    "Fromdate": "2024-02-01",
    "Todate": "2024-02-28",
    "Pickup": false,
    "Invoicecheck": false
}

// Filters WHERE:
// - CompanyRefId = 1
// - CustomerRefId = 5
// - JobMasterRefId = 2
// - EmployeeRefId = 3 (since DashboardStatus != 2, no subquery)
// - JStatus IN (1,2)
// - JStatus != 8
// - Remarks IS NOT NULL AND Remarks != ''
// - Offvesselname LIKE '%Maersk%'
// - ETA BETWEEN '2024-02-01 00:00:00' AND '2024-02-28 23:59:59'
// - Active = 1

// ============================================================================
// FIELD DESCRIPTIONS
// ============================================================================

/*
Filter Request Fields:
- Comid (Integer, REQUIRED): Company ID for data isolation
- Id (Integer): Customer Reference ID (optional filter)
- JId (Integer): Job Master Reference ID (optional filter)
- Employeeid (Integer): Employee Reference ID (optional filter)
- DashboardStatus (Integer):
  - 0 or 1: Simple employee filter
  - 2: Include subordinates via RulesTypeMaster subquery
- statusList (String): Comma-separated status IDs (e.g., "1,2,3")
- Statusid (Integer): Single status ID (ignored if statusList is provided)
- completestatusnotshow (Boolean): Exclude JStatus = 8 if true
- Remarks (Integer):
  - 0: No filter
  - 1: Show only orders with non-empty remarks
  - 2: Show only orders with empty remarks
- Offvesselname (String): Off-loading vessel name (LIKE search)
- Loadingvesselname (String): Loading vessel name (LIKE search)
- Search (String): Search bill/invoice number (overrides date filters)
- Invoice (Boolean):
  - false: Search in SaleOrderMaster.CNumberDisplay
  - true: Search in SaleMaster.CNumberDisplay
- ETA (Boolean): Enable ETA-based filtering
- ETAType (Integer):
  - 1: Filter by OETA (original ETA)
  - 2: Filter by ETA (estimated time of arrival)
  - other: Filter by both OETA OR ETA
- Fromdate (LocalDate): Start date for range filters (format: YYYY-MM-DD)
- Todate (LocalDate): End date for range filters (format: YYYY-MM-DD)
- Pickup (Boolean): Filter by PickupDate instead of default SaleDate
- Invoicecheck (Boolean): Show only orders with InvoiceNo = 0

Response Fields:
- IsSuccess (Boolean): Operation success indicator
- StatusCode (Integer): HTTP status code
- Message (String): Status message
- Data1 (SaleF5View): Actual response data (master + details)
- Data3 (String): JSON serialized copy of Data1
- ErrorDetails (String): Error details for debugging

SaleMasterViewModel Fields:
- Id: Order ID
- sportsaleorderid: Sports sale order ID
- InvoiceId: Invoice ID
- Remarks: Order remarks
- Destination: Destination port/location
- FlighTime: Flight time (if applicable)
- Origin: Origin port/location
- JobMasterRefId: Job type ID
- EmployeeName: Assigned employee name
- Offvesselname: Off-loading vessel name
- Sname: Currency symbol name
- Loadingvesselname: Loading vessel name
- SPort: Source/origin port
- OPort: Off-loading port
- BillDate: Formatted sale date (dd/MM/yyyy)
- DETA: Dynamic ETA (selected based on ETAType)
- ETA: Estimated time of arrival (LocalDateTime)
- SETA: Formatted ETA (dd/MM/yyyy HH:mm:ss)
- SETB: Formatted ETB (dd/MM/yyyy HH:mm:ss)
- SOETA: Formatted OETA (dd/MM/yyyy HH:mm:ss)
- SOETB: Formatted OETB (dd/MM/yyyy HH:mm:ss)
- SPickupDate: Formatted pickup date
- BillNoDisplay: Bill number for display
- BillTime: Bill creation time formatted
- CustomerName: Customer name
- JobType: Job type name
- NetAmt: Net amount
- SaleType: Sale type (e.g., International, Domestic)
- BillNo: Bill number (integer)
- JobStatus: Current job status
- InvoiceNo: Invoice number (string)
- QNECode: QNE code
- QNEId: QNE ID

SaleDetailsViewModel Fields:
- DiscountAmt: Discount amount
- DiscountPercent: Discount percentage
- ItemQty: Item quantity
- MRP: Maximum retail price
- ProductName: Product name
- SDRemarks: Sale detail remarks
- SaleRate: Sale rate per unit
- SaleRefId: Reference to SaleOrderMaster ID
- TaxAmt: Tax amount
- TaxPercent: Tax percentage
- ProductCode: Product code
- SAmount: Sale amount
- CurrencyValue: Currency value factor
- ActualAmount: Actual amount after currency conversion
*/
